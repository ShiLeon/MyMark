Task_1000ms{

    if(全局时间无效且没有进行过授时){

    	使用RTC授时(获取RTC时间：多通道I2C开关（tca9548）-->RTC模块（Ins5a8900）)

    }

    else if(PBOX超过全局时间10s){

    	使用PBOX授时，并且设置RTC时间

    }

}

}
